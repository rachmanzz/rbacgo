from pathlib import Path
import json,os,signal,subprocess,time

root=Path('/home/natadana/rahman-project/rbacgo/.knowledges/audit/verification_v7')
runtime=root/'runtime'
results=[]
for mode in ('barrier','continuous'):
    for round_no in range(10):
        path=runtime/f'{mode}-{round_no}.jsonl'
        assert not path.exists(), path
        writer=subprocess.Popen(['/tmp/rbacgo-audit-v7/crash-probe',mode,str(path)],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,bufsize=1)
        acknowledged=[]
        try:
            for number in range(20):
                line=writer.stdout.readline()
                if not line:
                    raise RuntimeError(writer.stderr.read())
                acknowledged.append(json.loads(line)['ack'])
                if mode=='barrier' and number<19:
                    writer.stdin.write('\n')
                    writer.stdin.flush()
            os.kill(writer.pid,signal.SIGKILL)
            writer.wait(timeout=10)
            assert writer.returncode == -signal.SIGKILL, writer.returncode
        finally:
            if writer.poll() is None:
                writer.kill()
                writer.wait(timeout=10)
            writer.stdin.close()
            writer.stdout.close()
            writer.stderr.close()
        raw=path.read_bytes()
        complete=raw[:raw.rfind(b'\n')+1]
        records=[json.loads(line) for line in complete.splitlines()]
        observed=[int(record['ID']) for record in records]
        assert len(observed)==len(set(observed)), 'duplicate event ID'
        assert set(acknowledged)<=set(observed), 'acknowledged event lost'
        assert observed==sorted(observed), 'events reordered'
        partial_tail=len(complete)!=len(raw)
        recovery_mode='incomplete' if partial_tail else 'resume'
        recovered=subprocess.run(['/tmp/rbacgo-audit-v7/crash-probe',recovery_mode,str(path)],capture_output=True,text=True,timeout=10,check=True)
        expected='REJECTED_UNCHANGED' if partial_tail else 'RECOVERED'
        assert recovered.stdout.strip()==expected,recovered.stdout
        results.append({'mode':mode,'round':round_no,'acknowledged':len(acknowledged),'complete_before_recovery':len(records),'incomplete_tail':partial_tail,'recovery':expected,'passed':True})
        path.unlink()
        if (round_no+1)%5==0:
            print(f'{mode}: {round_no+1}/10 SIGKILL rounds passed',flush=True)
# Deterministically exercise the incomplete-tail recovery contract as well.
partial=runtime/'incomplete.jsonl'
partial.write_bytes(b'{"ID":"prior","Operation":"complete"}\n{"ID":"partial"')
checked=subprocess.run(['/tmp/rbacgo-audit-v7/crash-probe','incomplete',str(partial)],capture_output=True,text=True,check=True)
assert checked.stdout.strip()=='REJECTED_UNCHANGED'
partial.unlink()
summary={'filesystem':'ext4 (findmnt verified)','process_crash_rounds':len(results),'acknowledged_events_checked':sum(item['acknowledged'] for item in results),'all_passed':True,'deterministic_incomplete_tail':'rejected without mutation','actual_power_loss_tested':False,'windows_runtime_tested':False,'rounds':results}
(root/'crash-results.json').write_text(json.dumps(summary,indent=2)+'\n')
print(f"PASS: {len(results)} SIGKILL rounds; {summary['acknowledged_events_checked']} acknowledged events recovered; incomplete tails fail closed",flush=True)
